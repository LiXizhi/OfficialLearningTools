-- Paracraft 代码方块 (CodeBlock) · 校门虚拟具身迎宾机器人 (g3-01)
-- 装配方式：拉杆 -> 代码方块 (219) <-> 电影方块 (含迎宾机器人演员)
-- 遵循 c:/lxzsrc/keepworkExtension/skills/paracraft-create/references/code-blocks.md 规范
setActorValue("sentientRadius", 0)
setActorValue("name", "gate_embodied_robot")

local mode = "normal" -- normal(正常清晰) | occluded(遮挡逆光) | noisy(环境嘈杂)

local function actByEnv(envMode)
    if envMode == "normal" then
        say("☀️【感知清晰 98%】认出本校同学校牌！同学早安，欢迎进入智慧校园！", 4)
        turnTo(0)
        walkForward(1, 0.8)
        wait(0.3)
        walkForward(-1, 0.8)
    elseif envMode == "occluded" then
        say("🕶️【视觉受阻 42%】镜头被遮挡看不清特征！我已停步亮黄灯，请摘下遮挡或请值班老师人工核验！", 5)
        turnTo(45)
    elseif envMode == "noisy" then
        say("📢【语音受扰】周围太吵把“下雪”听成“下学”啦！请走近麦克风重说一遍，或请老师帮忙确认！", 5)
        turnTo(-45)
    end
end

-- 启动时先执行一次正常迎宾
actByEnv(mode)

-- 点击具身机器人可切换控制变量实验条件
registerClickEvent(function()
    local idx = ask("🤖 我是校门虚拟具身迎宾机器人！请选择要测试的环境条件（控制变量实验）：", {
        "☀️ 1. 环境正常（画面清晰 + 声音清楚）",
        "🕶️ 2. 视觉干扰（雨伞/逆光遮挡摄像头一半）",
        "📢 3. 语音干扰（周围大声喧哗 + 同音字）"
    })
    if idx == 1 then
        mode = "normal"
    elseif idx == 2 then
        mode = "occluded"
    elseif idx == 3 then
        mode = "noisy"
    else
        return
    end
    actByEnv(mode)
end)

校门口的两次停靠
在迎宾机器人的世界入口加一块观察牌。请访客拿清楚和遮住的星卡各试一次，再去按电灯开关；看他能不能指出谁需要辨认图案。


  
桌上的材料
  • 星卡 A 画着完整的五角星；模拟识别器举起“看清了”牌。 
  • 星卡 B 是同一张卡，只盖住半颗星；模拟识别器举起“看不清”牌。 
  • 值班台在入口左边。普通电灯开关只在按下时通电，旁边放什么图案都一样。 
  • 拓展展位有一个识花助手，它需要花瓣图像；手摇铃只靠人摇。 
使用约定　只用虚构纸卡模拟感知，不拍脸、不登记姓名；访客身份由值班老师核实。

行为脚本
local tests = {
    {q="摆上星卡 A 后，哪个动作体现了机器人在处理输入？", options={"先辨认图案再欢迎", "每次都通电亮灯", "随风摆动"}, correct=1, evidence="材料 A 给的是完整星形，模拟器先报告看清，再由机器人欢迎。"},
    {q="只把星卡遮半，什么动作最符合试验约定？", options={"当作另一位同学", "停在入口，请去左边值班台", "自动登记访客"}, correct=2, evidence="材料 B 给出看不清，值班台在左边，没有确认访客身份的依据。"},
    {q="把入口换成花园，哪个设备也要先从输入里找特征？", options={"手摇铃", "按键灯", "根据花瓣图像识花的助手"}, correct=3, evidence="拓展展位写明识花助手要看花瓣；另两件由手直接推动或按动。"}
}
setActorValue("sentientRadius", 0)
registerClickEvent(function()
    local index = ask("想复核哪一次测试？", {"基准", "改变条件", "换个场景"})
    local test = tests[index]
    if not test then return end
    local answer = ask(test.q, test.options)
    if not answer then return end
    if answer == test.correct then
        say("判断成立。" .. test.evidence, 6)
